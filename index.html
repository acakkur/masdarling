Arsitektur Cloudflare Full-Stack & Skema D1

100% Ekosistem Cloudflare (Pages, Workers, D1, R2, Queues, Cron, Turnstile) + Google Sheets Sync

1. Cloudflare Workers + D1 SQLite

Seluruh query menggunakan prepared statements & transaksi atomik. Akses database hanya via Worker API dengan RBAC & Region Guard (RW 01–05).

2. Cloudflare R2 & Queues

Bukti bayar & kuitansi disimpan pada R2 private bucket dengan randomized UUID key & signed URL berumur pendek. Sinkronisasi Sheets berjalan asinkron via Queue.

-- RINGKASAN TABEL UTAMA CLOUDFLARE D1 (migrations/0001_init.sql)
PRAGMA foreign_keys = ON;

CREATE TABLE rw_regions (id TEXT PRIMARY KEY, code TEXT UNIQUE NOT NULL, name TEXT NOT NULL);
CREATE TABLE rt_regions (id TEXT PRIMARY KEY, rw_id TEXT NOT NULL REFERENCES rw_regions(id), code TEXT NOT NULL);
CREATE TABLE users (id TEXT PRIMARY KEY, username TEXT UNIQUE NOT NULL, password_hash TEXT NOT NULL, role TEXT CHECK(role IN ('ADMIN','PETUGAS')) NOT NULL, is_active INTEGER DEFAULT 1);
CREATE TABLE officer_assignments (id TEXT PRIMARY KEY, user_id TEXT NOT NULL REFERENCES users(id), rw_id TEXT NOT NULL REFERENCES rw_regions(id), UNIQUE(user_id, rw_id));
CREATE TABLE residents (id TEXT PRIMARY KEY, customer_code TEXT UNIQUE NOT NULL, full_name TEXT NOT NULL, address TEXT NOT NULL, rt_id TEXT NOT NULL, rw_id TEXT NOT NULL, category TEXT NOT NULL, monthly_tariff INTEGER NOT NULL, status TEXT DEFAULT 'ACTIVE', deleted_at TEXT);
CREATE TABLE invoices (id TEXT PRIMARY KEY, invoice_number TEXT UNIQUE NOT NULL, resident_id TEXT NOT NULL REFERENCES residents(id), rw_id TEXT NOT NULL, period_month INTEGER NOT NULL, period_year INTEGER NOT NULL, amount INTEGER NOT NULL, status TEXT CHECK(status IN ('UNPAID','PENDING','PAID','REJECTED','CANCELLED')) NOT NULL, UNIQUE(resident_id, period_month, period_year));
CREATE TABLE payments (id TEXT PRIMARY KEY, tx_number TEXT UNIQUE NOT NULL, invoice_id TEXT NOT NULL REFERENCES invoices(id), resident_id TEXT NOT NULL, rw_id TEXT NOT NULL, amount INTEGER NOT NULL, method TEXT NOT NULL, status TEXT CHECK(status IN ('PENDING','SUBMITTED','VERIFIED','REJECTED','CANCELLED')) NOT NULL, receipt_r2_key TEXT, idempotency_key TEXT UNIQUE NOT NULL);
CREATE TABLE ledger_entries (id TEXT PRIMARY KEY, entry_type TEXT CHECK(entry_type IN ('CREDIT','DEBIT','ADJUSTMENT')) NOT NULL, ref_type TEXT NOT NULL, ref_id TEXT NOT NULL, amount INTEGER NOT NULL, balance_after INTEGER NOT NULL, created_by TEXT NOT NULL, created_at TEXT NOT NULL);
CREATE TABLE audit_logs (id TEXT PRIMARY KEY, actor_id TEXT, action TEXT NOT NULL, entity_type TEXT NOT NULL, entity_id TEXT NOT NULL, payload_json TEXT, created_at TEXT NOT NULL);
